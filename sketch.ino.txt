#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include <WiFi.h>
#include <HTTPClient.h> 

// Hardware Mapping matching your exact canvas layout
#define LED_GREEN     ((gpio_num_t)12)
#define LED_YELLOW    ((gpio_num_t)14)
#define LED_RED       ((gpio_num_t)27)
#define BUZZER        ((gpio_num_t)26)
#define SWITCH_PIN    ((gpio_num_t)5)

// ❗ UPDATE THIS LINK WITH YOUR LAPTOP'S REAL IP ADDRESS
const char* python_gateway_url = "http://192.168.178.141:8000/"; 

volatile int current_safety_tier = 0; 
volatile float current_temperature = 24.0;

void init_hardware() {
    gpio_reset_pin(LED_GREEN);  gpio_set_direction(LED_GREEN, GPIO_MODE_OUTPUT);
    gpio_reset_pin(LED_YELLOW); gpio_set_direction(LED_YELLOW, GPIO_MODE_OUTPUT);
    gpio_reset_pin(LED_RED);    gpio_set_direction(LED_RED, GPIO_MODE_OUTPUT);
    gpio_reset_pin(BUZZER);     gpio_set_direction(BUZZER, GPIO_MODE_OUTPUT);
    
    gpio_reset_pin(SWITCH_PIN);
    gpio_set_direction(SWITCH_PIN, GPIO_MODE_INPUT);
    gpio_set_pull_mode(SWITCH_PIN, GPIO_PULLUP_ONLY);
}

// Thread 1: Handles real-time physical hardware safety priorities
void vHardwareControlTask(void *pvParameters) {
    while(1) {
        // 1. Highest Priority Override: Emergency Thermal State via Slide Switch
        if (current_temperature > 50.0) {
            printf("[OVERRIDE] CRITICAL HARDWARE EMERGENCY DETECTED!\n");
            gpio_set_level(LED_GREEN, 0); gpio_set_level(LED_YELLOW, 0); gpio_set_level(LED_RED, 1);
            
            // Pulse the alarm buzzer sharply
            gpio_set_level(BUZZER, 1); vTaskDelay(pdMS_TO_TICKS(100));
            gpio_set_level(BUZZER, 0); vTaskDelay(pdMS_TO_TICKS(100));
            continue;
        }

        // 2. Standard Operation Tiers driven asynchronously by Python AI Network Link
        switch(current_safety_tier) {
            case 0: // Safe State
                gpio_set_level(LED_GREEN, 1); gpio_set_level(LED_YELLOW, 0); gpio_set_level(LED_RED, 0); gpio_set_level(BUZZER, 0);
                vTaskDelay(pdMS_TO_TICKS(100)); break;
                
            case 1: // Warning State (Blink Yellow)
                gpio_set_level(LED_GREEN, 0); gpio_set_level(LED_RED, 0); gpio_set_level(BUZZER, 0);
                gpio_set_level(LED_YELLOW, 1); vTaskDelay(pdMS_TO_TICKS(200));
                gpio_set_level(LED_YELLOW, 0); vTaskDelay(pdMS_TO_TICKS(200)); break;
                
            case 2: // Critical Absence State (Lock Red + Sound Buzzer)
                gpio_set_level(LED_GREEN, 0); gpio_set_level(LED_YELLOW, 0); gpio_set_level(LED_RED, 1);
                gpio_set_level(BUZZER, 1); vTaskDelay(pdMS_TO_TICKS(150));
                gpio_set_level(BUZZER, 0); vTaskDelay(pdMS_TO_TICKS(150)); break;
        }
    }
}

// Thread 2: Monitors physical plant sensors concurrently
void vEnvironmentMonitoringTask(void *pvParameters) {
    while(1) {
        if (gpio_get_level(SWITCH_PIN) == 0) {
            current_temperature = 55.0; // Fail-safe trigger
        } else {
            current_temperature = 24.0; 
        }
        vTaskDelay(pdMS_TO_TICKS(200)); 
    }
}

void setup() {
    init_hardware();
    Serial.begin(115200);
    
    // Connect to Wokwi's free network bridge
    printf("[NETWORK] Initializing WiFi Connection to Wokwi-GUEST...\n");
    WiFi.begin("Wokwi-GUEST", "", 6);
    while (WiFi.status() != WL_CONNECTED) { 
        vTaskDelay(pdMS_TO_TICKS(500)); 
    }
    printf("[NETWORK] Wireless Bridge Established Successfully!\n");
    
    // Launch our parallel execution architecture across the CPU cores
    xTaskCreate(vHardwareControlTask, "HardwareControl", 2048, NULL, 5, NULL);
    xTaskCreate(vEnvironmentMonitoringTask, "EnvironmentMonitoring", 2048, NULL, 4, NULL);
}

// Background thread: Continuously fetches data from your Python script over HTTP
void loop() {
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        http.begin(python_gateway_url);
        
        int httpResponseCode = http.GET();
        if (httpResponseCode > 0) {
            String payload = http.getString();
            
            if (payload == "TIER_0") current_safety_tier = 0;
            else if (payload == "TIER_1") current_safety_tier = 1;
            else if (payload == "TIER_2") current_safety_tier = 2;
        }
        http.end();
    }
    vTaskDelay(pdMS_TO_TICKS(1000)); // Query the Python AI status every 1 second
}
