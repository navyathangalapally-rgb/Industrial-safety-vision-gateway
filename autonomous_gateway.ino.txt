#include <WiFi.h>
#include <HTTPClient.h>
#include "hardware_config.h"
#include "safety_tiers.h"

const char* python_gateway_url = "http://192.168.178.141:8000/";

volatile int current_safety_tier = 0;
volatile float current_temperature = 24.0;

void vHardwareControlTask(void *pvParameters) {
    while(1) { handle_safety_execution(); }
}

void vEnvironmentMonitoringTask(void *pvParameters) {
    while(1) {
        current_temperature = (gpio_get_level(SWITCH_PIN) == 0) ? 55.0 : 24.0;
        vTaskDelay(pdMS_TO_TICKS(200));
    }
}

void setup() {
    init_hardware_pins();
    Serial.begin(115200);
    
    WiFi.begin("Wokwi-GUEST", "", 6);
    while (WiFi.status() != WL_CONNECTED) { vTaskDelay(pdMS_TO_TICKS(500)); }
    
    xTaskCreate(vHardwareControlTask, "HardwareControl", 2048, NULL, 5, NULL);
    xTaskCreate(vEnvironmentMonitoringTask, "EnvironmentMonitoring", 2048, NULL, 4, NULL);
}

void loop() {
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        http.begin(python_gateway_url);
        int httpResponseCode = http.GET();
        if (httpResponseCode > 0) {
            String payload = http.getString();
            if (payload == "TIER_0") current_safety_tier = 0;
            else if (payload == "TIER_1") current_safety_tier = 1;
            else if (payload == "TIER_2") current_safety_tier = 2;
        }
        http.end();
    }
    vTaskDelay(pdMS_TO_TICKS(1000));
}
